// 查看可用的区域

// 预览创建集群
kops create cluster --cloud=aws --zones=ap-southeast-1a --name=$NAME --dns-zone=$NAME --dns private
--cloud		云服务器提供商
--zones 	集群所在的区域，多个以逗号分隔
--name		集群名称
--state		集群状态的S3存储桶
--yes		立即创建集群

// 生成Terraform文件
kops create cluster --cloud=aws --zones=ap-southeast-1a --name=$NAME --dns-zone=$NAME --dns private --out=. --target=terraform

// 创建集群
kops update cluster $NAME --yes

// 编辑集群
kops edit cluster $NAME

// 当确定要删除群集时，请发出带有--yes标志的delete命令。请注意，此命令具有很大的破坏性，它将删除您的集群及其中包含的所有内容
kops delete cluster --name ${NAME} --yes

# 从模板生成cluster.yaml
kops toolbox template \
    --values values.yaml --values=another.yaml \
    --set var=value --set-string othervar=true \
    --snippets file_or_directory --snippets=another.dir \
    --template file_or_directory --template=directory  \
    --output cluster.yaml
    
# 实际使用中，会结合Terraform的输出生成cluster.yaml
TF_OUTPUT=$(cd ../terraform && terraform output -json)
CLUSTER_NAME="$(echo ${TF_OUTPUT} | jq -r .cluster_name.value)"
STATE="s3://$(echo ${TF_OUTPUT} | jq -r .kops_s3_bucket_name.value)"
kops toolbox template --name ${CLUSTER_NAME} --values <( echo ${TF_OUTPUT}) --template template.yaml --format-yaml > cluster.yaml

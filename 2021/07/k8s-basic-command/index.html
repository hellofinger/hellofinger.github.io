# 创建
kubectl run <deployment> --image=<image> --replicas=<num>

# 升级
kubectl set image <deploymnet> nginx=nginx:1.9.1
kubectl set resources <deployment> -c =nginx --limits=cpu=200m,memory=512Mi

# 暂停
kubectl rollout pause <deployment>

# 恢复
kubectl rollout resume <deployment>

# 查询升级状态
kubectl rollout status <deployment>

# 查询升级历史
kubectl rollout history <deployment>
kubectl rollout history <deployment> - revision=2

# 回滚
kubectl rollout undo <deployment> --to-revision=2

# 弹性伸缩
kubectl scale deployment <deployment> --replicas=10
kubectl autoscale deployment <deployment> --min=10 --max=15 --cpu-percent=80
